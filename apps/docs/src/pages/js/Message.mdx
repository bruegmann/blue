# Message

These functions are meant as a nicer alternative to the native `alert`, `confirm` and `prompt` functions.
They are not meant to be used as a replacement for more complex modals.

## Functions

### Verify

```html use:RenderBlock
<button type="button" class="btn" onclick="verifyUser()">Verify</button>

<script>
    async function verifyUser() {
        const yesOrNo = await blueWeb.message.verify("Do you want to continue?")
        await blueWeb.message.tell(yesOrNo ? "You clicked yes" : "You clicked no")
    }
</script>
```

### Tell

```html use:RenderBlock
<button type="button" class="btn" onclick="tellUser()">Tell</button>

<script>
    async function tellUser() {
        await blueWeb.message.tell("A message for you")
    }
</script>
```

### Ask

```html use:RenderBlock
<button type="button" class="btn" onclick="askUser()">Ask</button>

<script>
    async function askUser() {
        const answer = await blueWeb.message.ask("What is the question?", {
            title: "Question"
        })
        await blueWeb.message.tell("You asked: " + answer)
    }
</script>
```

### Ask with default value

```html use:RenderBlock
<button type="button" class="btn" onclick="askUserWithDefaultValue()">Ask</button>

<script>
    async function askUserWithDefaultValue() {
        const answer = await blueWeb.message.ask(
            "What is your name?",
            `{"title":"Save Query","defaultValue":"DEFAULT LGK"}`
        )
        await blueWeb.message.tell("You asked: " + answer)
    }
</script>
```

### Control dialog by ID

```html use:RenderBlock
<button type="button" class="btn" onclick="openControlledTellDialog()">Open controlled dialog</button>

<script>
    async function openControlledTellDialog() {
        setTimeout(() => {
            document.getElementById("myControlledDialog").close()
        }, 3000)
        await blueWeb.message.tell("Dialog will close after a few seconds.", { id: "myControlledDialog" })
    }
</script>
```
